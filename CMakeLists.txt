cmake_minimum_required(VERSION 3.20)
project(OrraBasin VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
include(CTest)
option(ORRA_COVERAGE "Instrument first-party C for gcov" OFF)
set(ORRA_SDL_SOURCE "" CACHE PATH "Optional SDL3 source tree")
set(ORRA_LUA_SOURCE "" CACHE PATH "Optional Lua 5.4 source tree")
if(NOT ORRA_SDL_SOURCE AND (NOT EMSCRIPTEN OR SDL3_DIR))
  find_package(SDL3 3.2 QUIET CONFIG)
endif()
if(NOT TARGET SDL3::SDL3)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  if(ORRA_SDL_SOURCE)
    add_subdirectory("${ORRA_SDL_SOURCE}" sdl EXCLUDE_FROM_ALL)
  else()
    FetchContent_Declare(sdl URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-3.2.26
      URL_HASH SHA256=73eeeec0d3235ef75f310bd2bb758622157abfa05da65771656579a7dfeeba05)
    FetchContent_MakeAvailable(sdl)
  endif()
endif()
if(NOT ORRA_LUA_SOURCE)
  FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
    URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae)
  FetchContent_MakeAvailable(lua)
  set(ORRA_LUA_SOURCE "${lua_SOURCE_DIR}")
endif()
file(GLOB LUA_C "${ORRA_LUA_SOURCE}/src/*.c")
list(FILTER LUA_C EXCLUDE REGEX "/(lua|luac|onelua)\\.c$")
add_library(orra_lua STATIC ${LUA_C})
if(UNIX AND NOT EMSCRIPTEN)
  target_compile_definitions(orra_lua PRIVATE LUA_USE_POSIX)
endif()
target_include_directories(orra_lua PUBLIC "${ORRA_LUA_SOURCE}/src")
add_library(engine STATIC src/sim.c src/world.c src/render.c src/ui.c src/app.c src/xr.c src/replay.c src/main.c)
target_include_directories(engine PUBLIC src)
target_link_libraries(engine PUBLIC SDL3::SDL3 orra_lua)
if(NOT MSVC)
  target_link_libraries(engine PUBLIC m)
  target_compile_options(engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ORRA_COVERAGE)
  target_compile_options(engine PRIVATE --coverage -O0 -g)
  target_link_options(engine INTERFACE --coverage)
endif()
add_executable(1stprts src/entry.c)
target_link_libraries(1stprts PRIVATE engine)
if(ORRA_COVERAGE)
  target_compile_options(1stprts PRIVATE --coverage -O0 -g)
endif()
set_target_properties(1stprts PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" upper)
  set_target_properties(1stprts PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(WIN32 AND TARGET SDL3::SDL3-shared)
  add_custom_command(TARGET 1stprts POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_FILE:SDL3::SDL3-shared>" "$<TARGET_FILE_DIR:1stprts>")
endif()
# Content is deployed as ordinary editable sibling files on every platform.
file(GLOB_RECURSE GAME_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_SOURCE_DIR}"
  "assets/*.mesh" "assets/*.txt" "assets/*.md" "assets/*.glsl" "assets/*.wav" "scripts/*.lua")
set(JSON_FILES "")
foreach(f IN LISTS GAME_FILES)
  configure_file("${f}" "${CMAKE_BINARY_DIR}/bin/${f}" COPYONLY)
  string(APPEND JSON_FILES "\"${f}\",")
endforeach()
string(REGEX REPLACE ",$" "" JSON_FILES "${JSON_FILES}")
file(WRITE "${CMAKE_BINARY_DIR}/bin/content.json" "{\"version\":1,\"files\":[${JSON_FILES}]}\n")
configure_file(LICENSE "${CMAKE_BINARY_DIR}/bin/LICENSE" COPYONLY)
foreach(notice CC-BY-4.0 SDL3 Lua)
  configure_file("LICENSES/${notice}.txt" "${CMAKE_BINARY_DIR}/bin/LICENSES/${notice}.txt" COPYONLY)
endforeach()
if(EMSCRIPTEN)
  target_link_options(1stprts PRIVATE -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sALLOW_MEMORY_GROWTH=1
    -sFORCE_FILESYSTEM=1 -sGL_ENABLE_GET_PROC_ADDRESS=1
    "-sEXPORTED_FUNCTIONS=['_main','_orra_status','_orra_xr_mode','_orra_xr_tick','_orra_xr_eye','_orra_xr_action','_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['FS','addRunDependency','removeRunDependency','GL','HEAPF32','UTF8ToString']")
  set_target_properties(1stprts PROPERTIES OUTPUT_NAME orra SUFFIX ".js")
  foreach(file index.html boot.js)
    configure_file("web/${file}" "${CMAKE_BINARY_DIR}/bin/${file}" COPYONLY)
  endforeach()
elseif(BUILD_TESTING)
  add_executable(unit_tests tests/unit.c tests/components.c)
  target_link_libraries(unit_tests PRIVATE engine)
  add_test(NAME cli_units COMMAND 1stprts --help)
  add_test(NAME engine_units COMMAND unit_tests "${CMAKE_BINARY_DIR}/bin")
  add_executable(lua_tests tests/lua_host.c)
  target_link_libraries(lua_tests PRIVATE orra_lua)
  if(NOT MSVC)
    target_link_libraries(lua_tests PRIVATE m)
  endif()
  add_test(NAME lua_units COMMAND lua_tests "${CMAKE_SOURCE_DIR}/tests/rules.lua" "${CMAKE_BINARY_DIR}/bin")
  add_test(NAME gate_six_replay COMMAND 1stprts --replay "${CMAKE_SOURCE_DIR}/tests/gate-six.route")
  add_test(NAME rival_replay COMMAND 1stprts --replay "${CMAKE_SOURCE_DIR}/tests/rival.route")
  add_test(NAME armor_first_replay COMMAND 1stprts --replay "${CMAKE_SOURCE_DIR}/tests/armor-first.route")
  add_test(NAME dispatch_replay COMMAND 1stprts --replay "${CMAKE_SOURCE_DIR}/tests/dispatch.route")
  add_test(NAME gantry_replay COMMAND 1stprts --replay "${CMAKE_SOURCE_DIR}/tests/gantry.route")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_tests_properties(gate_six_replay rival_replay armor_first_replay dispatch_replay gantry_replay engine_units PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=offscreen;SDL_AUDIODRIVER=dummy;LIBGL_ALWAYS_SOFTWARE=1")
  endif()
endif()
